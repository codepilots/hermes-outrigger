# Test + runtime dependencies for the outrigger plugin suite.
# Runtime dep is declared in outrigger/plugin.yaml (jsonschema>=4,<5); aiohttp
# is provided by the Hermes gateway. These pins make the suite reproducible
# from a fresh venv: python -m venv .venv && .venv/bin/pip install -r requirements-dev.txt
# Ranges track the versions the suite is validated on (pytest 9.2.0,
# pytest-asyncio 1.5.0, pytest-aiohttp 1.1.1, aiohttp 3.14.5, jsonschema 4.26.0);
# the earlier pytest<9 / pytest-asyncio<1 pins predated the validated dev venv.
pytest>=9,<10
pytest-asyncio>=1,<2
pytest-aiohttp>=1.0,<2
aiohttp>=3.9,<4
jsonschema>=4,<5
